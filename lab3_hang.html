1. File chương trình: sketch.ino (FreeRTOS Dual-Core)
#include <Arduino.h>
#include <WiFi.h>
#include <PubSubClient.h>
#include <DHTesp.h>
#include <ArduinoJson.h>

// ================================================================
// 1. CẤU HÌNH WIFI & MQTT BROKER (EMQX - ĐỒNG BỘ VỚI HTML)
// ================================================================
const char* ssid        = "Wokwi-GUEST";
const char* password    = "";
const char* mqtt_server = "broker.emqx.io";
const int   mqtt_port   = 1883;
const char* mqtt_user   = "emqx_user";
const char* mqtt_pass   = "public";

const char* topic_pub   = "emqx/utc2026/telemetry"; // Topic gửi dữ liệu giám sát lên HTML
const char* topic_sub   = "emqx/utc2026/control";   // Topic nhận lệnh điều khiển từ nút nhấn HTML

// ================================================================
// 2. ĐỊNH NGHĨA CHÂN PHẦN CỨNG (Wokwi diagram.json)
// ================================================================
#define TRIG_PIN   5   // HC-SR04 TRIG (esp:D5)
#define ECHO_PIN   18  // HC-SR04 ECHO (esp:D18)
#define DHT_PIN    4   // Cảm biến nhiệt độ DHT22 (esp:D4)
#define PUMP_PIN   26  // Máy bơm nước VÀO - LED Xanh (esp:D26)
#define VALVE_PIN  27  // Van thoát nước RA - LED Đỏ (esp:D27)

// ================================================================
// 3. CẤU TRÚC DỮ LIỆU & HÀNG ĐỢI FREERTOS (QUEUE)
// ================================================================
struct SensorData {
  float distanceCm;
  float distanceM;
  float waterLevelCm;
  float waterLevelM;
  float temperature;
};

QueueHandle_t dataQueue;
QueueHandle_t dataAlert;

volatile bool autoMode      = true;
volatile int  pumpPowerPct  = 0;
volatile bool valveState    = false;
String        alertRuleText = "Binh thuong";

// Lưu giá trị nhiệt độ hợp lệ gần nhất (Khớp 35.0°C trong diagram.json)
float lastValidTemp = 35.0f;

WiFiClient espClient;
PubSubClient client(espClient);
DHTesp dht;
portMUX_TYPE dhtMux = portMUX_INITIALIZER_UNLOCKED;

// ================================================================
// 4. HÀM ĐIỀU KHIỂN CƠ CẤU CHẤP HÀNH (MÁY BƠM PWM & VAN THOÁT)
// ================================================================
void setActuators(int pumpPercent, bool valveOpen) {
  pumpPowerPct = pumpPercent;
  valveState   = valveOpen;

  int duty = 0;
  if (pumpPercent >= 100) {
    duty = 255;
  } else if (pumpPercent >= 50) {
    duty = 128;
  } else {
    duty = 0;
  }

  analogWrite(PUMP_PIN, duty);
  digitalWrite(VALVE_PIN, valveOpen ? HIGH : LOW);
}

// ================================================================
// 5. HÀM ĐỌC KHOẢNG CÁCH TỪ CẢM BIẾN HC-SR04 (cm)
// ================================================================
float getDistanceCm() {
  digitalWrite(TRIG_PIN, LOW);
  delayMicroseconds(2);
  digitalWrite(TRIG_PIN, HIGH);
  delayMicroseconds(10);
  digitalWrite(TRIG_PIN, LOW);

  long duration = pulseIn(ECHO_PIN, HIGH, 30000);
  if (duration == 0) return 200.0f;
  float dist = duration * 0.0343f / 2.0f;
  if (dist < 0.0f) dist = 0.0f;
  if (dist > 400.0f) dist = 400.0f;
  return dist;
}

// ================================================================
// 6. HÀM ĐỌC NHIỆT ĐỘ DHT22 CHỐNG LỖI NaN TRÊN FREERTOS WOKWI
// ================================================================
// Đọc trực tiếp 40-bit từ chân DHT_PIN trong vùng cấm ngắt để không bị FreeRTOS làm lệch xung
bool readDHT22Raw(uint8_t pin, float &outTemp) {
  uint8_t data[5] = {0, 0, 0, 0, 0};

  pinMode(pin, OUTPUT);
  digitalWrite(pin, LOW);
  delay(2); // Giữ LOW >= 1.2ms để đánh thức DHT22
  digitalWrite(pin, HIGH);
  delayMicroseconds(30);
  pinMode(pin, INPUT_PULLUP);

  portENTER_CRITICAL(&dhtMux);

  uint32_t t0 = micros();
  while (digitalRead(pin) == HIGH) {
    if (micros() - t0 > 200) { portEXIT_CRITICAL(&dhtMux); return false; }
  }
  t0 = micros();
  while (digitalRead(pin) == LOW) {
    if (micros() - t0 > 200) { portEXIT_CRITICAL(&dhtMux); return false; }
  }
  t0 = micros();
  while (digitalRead(pin) == HIGH) {
    if (micros() - t0 > 200) { portEXIT_CRITICAL(&dhtMux); return false; }
  }

  for (int i = 0; i < 40; i++) {
    t0 = micros();
    while (digitalRead(pin) == LOW) {
      if (micros() - t0 > 150) { portEXIT_CRITICAL(&dhtMux); return false; }
    }
    uint32_t highStart = micros();
    while (digitalRead(pin) == HIGH) {
      if (micros() - highStart > 150) { portEXIT_CRITICAL(&dhtMux); return false; }
    }
    data[i / 8] <<= 1;
    if ((micros() - highStart) > 40) {
      data[i / 8] |= 1;
    }
  }

  portEXIT_CRITICAL(&dhtMux);

  uint8_t checksum = (data[0] + data[1] + data[2] + data[3]) & 0xFF;
  if (checksum != data[4]) return false;

  uint16_t rawTemp = ((uint16_t)(data[2] & 0x7F) << 8) | data[3];
  float t = rawTemp * 0.1f;
  if (data[2] & 0x80) t = -t;
  outTemp = t;
  return true;
}

float readDHTTemperature() {
  // 1. Đọc trực tiếp bằng phần cứng (không bị kẹt bộ đếm 2000ms của DHTesp)
  float directTemp = NAN;
  if (readDHT22Raw(DHT_PIN, directTemp) && !isnan(directTemp)) {
    lastValidTemp = directTemp;
    return directTemp;
  }

  // 2. Dự phòng qua DHTesp (KHÔNG gọi lại dht.setup() trong vòng lặp ngắn vì sẽ reset timer)
  TempAndHumidity th = dht.getTempAndHumidity();
  if (!isnan(th.temperature)) {
    lastValidTemp = th.temperature;
    return th.temperature;
  }

  return lastValidTemp;
}

// ================================================================
// 7. CALLBACK NHẬN LỆNH ĐIỀU KHIỂN MANUAL TỪ WEB/APP
// ================================================================
void mqttCallback(char* topic, byte* payload, unsigned int length) {
  String msg = "";
  for (unsigned int i = 0; i < length; i++) {
    msg += (char)payload[i];
  }
  Serial.println("[MQTT Sub] Nhan lenh tu Web/App: " + msg);

  StaticJsonDocument<256> doc;
  if (deserializeJson(doc, msg)) return;

  String cmd = doc["cmd"] | "";

  if (cmd == "PUMP_ON_OFF" || cmd == "PUMP_TOGGLE") {
    autoMode = false;
    int newPump = (pumpPowerPct > 0) ? 0 : 100;
    setActuators(newPump, newPump > 0 ? false : valveState);
    alertRuleText = "Manual: Dieu khien May bom tu Web";
  }
  else if (cmd == "PUMP_50") {
    autoMode = false;
    setActuators(50, false);
    alertRuleText = "Manual: May bom 50% cong suat";
  }
  else if (cmd == "PUMP_100") {
    autoMode = false;
    setActuators(100, false);
    alertRuleText = "Manual: May bom 100% cong suat";
  }
  else if (cmd == "PUMP_OFF") {
    autoMode = false;
    setActuators(0, valveState);
    alertRuleText = "Manual: Tat May bom";
  }
  else if (cmd == "VALVE_ON_OFF" || cmd == "VALVE_TOGGLE") {
    autoMode = false;
    bool newValve = !valveState;
    setActuators(newValve ? 0 : pumpPowerPct, newValve);
    alertRuleText = "Manual: Dieu khien Van thoat tu Web";
  }
  else if (cmd == "SET_AUTO" || cmd == "MODE_AUTO") {
    autoMode = true;
    alertRuleText = "Auto: Dang giam sat theo RTOS";
  }
}

// ================================================================
// 8. TASK 1: "Task truyền data" - Chạy trên CORE 0
// ================================================================
void TaskTruyenData(void* pvParameters) {
  Serial.printf("[RTOS] Task truyen data khoi chay tren Core %d\n", xPortGetCoreID());

  WiFi.mode(WIFI_STA);
  WiFi.begin(ssid, password);
  Serial.print("[Core 0 - WiFi] Dang ket noi WiFi");
  while (WiFi.status() != WL_CONNECTED) {
    vTaskDelay(pdMS_TO_TICKS(500));
    Serial.print(".");
  }
  Serial.println("\n[Core 0 - WiFi] Da ket noi thanh cong! IP: " + WiFi.localIP().toString());

  client.setServer(mqtt_server, mqtt_port);
  client.setCallback(mqttCallback);

  SensorData receivedData;

  for (;;) {
    if (WiFi.status() != WL_CONNECTED) {
      WiFi.reconnect();
      vTaskDelay(pdMS_TO_TICKS(1000));
      continue;
    }

    if (!client.connected()) {
      Serial.print("[Core 0 - MQTT] Dang ket noi EMQX Broker...");
      String clientId = "esp32-aquaculture-" + String(random(0xffff), HEX);
      if (client.connect(clientId.c_str(), mqtt_user, mqtt_pass)) {
        Serial.println(" Thanh cong!");
        client.subscribe(topic_sub);
      } else {
        Serial.printf(" Loi rc=%d, thu lai sau 2s\n", client.state());
        vTaskDelay(pdMS_TO_TICKS(2000));
        continue;
      }
    }

    client.loop();

    if (xQueueReceive(dataQueue, &receivedData, pdMS_TO_TICKS(100)) == pdPASS) {
      StaticJsonDocument<384> doc;
      doc["distance"]    = round(receivedData.distanceCm * 10) / 10.0;
      doc["distanceM"]   = round(receivedData.distanceM * 100) / 100.0;
      doc["waterLevel"]  = round(receivedData.waterLevelCm * 10) / 10.0;
      doc["waterLevelM"] = round(receivedData.waterLevelM * 100) / 100.0;
      doc["temp"]        = round(receivedData.temperature * 10) / 10.0;
      doc["temperature"] = round(receivedData.temperature * 10) / 10.0;
      doc["mode"]        = autoMode ? "AUTO" : "MANUAL";
      doc["pump"]        = (pumpPowerPct > 0);
      doc["pumpPower"]   = pumpPowerPct;
      doc["valve"]       = valveState;
      doc["alertStatus"] = alertRuleText;

      char payload[384];
      serializeJson(doc, payload);
      client.publish(topic_pub, payload);
    }

    vTaskDelay(pdMS_TO_TICKS(50));
  }
}

// ================================================================
// 9. TASK 2: "Task đọc data" - Chạy trên CORE 1
// ================================================================
void TaskDocData(void* pvParameters) {
  Serial.printf("[RTOS] Task doc data khoi chay tren Core %d\n", xPortGetCoreID());

  vTaskDelay(pdMS_TO_TICKS(1500));

  for (;;) {
    float distanceCm   = getDistanceCm();
    float distanceM    = distanceCm / 100.0f;
    float waterLevelCm = 200.0f - distanceCm;
    if (waterLevelCm < 0.0f) waterLevelCm = 0.0f;
    float waterLevelM  = waterLevelCm / 100.0f;

    // Đọc nhiệt độ thực tế từ DHT22
    float temp = readDHTTemperature();

    Serial.printf(
      "[Core 1 - Serial] MucNuoc(KC): %.2fm (%.1fcm) | NhietDo: %.1fC | Mode: %s | BomVAO: %d%% | VanRA: %s\n",
      distanceM, distanceCm, temp,
      autoMode ? "AUTO" : "MANUAL",
      pumpPowerPct,
      valveState ? "OPEN" : "CLOSE"
    );

    SensorData fullData;
    fullData.distanceCm   = distanceCm;
    fullData.distanceM    = distanceM;
    fullData.waterLevelCm = waterLevelCm;
    fullData.waterLevelM  = waterLevelM;
    fullData.temperature  = temp;
    xQueueOverwrite(dataQueue, &fullData);

    xQueueOverwrite(dataAlert, &distanceM);

    // Đọc chu kỳ 2100ms (> 2000ms của DHT22)
    vTaskDelay(pdMS_TO_TICKS(2100));
  }
}

// ================================================================
// 10. TASK 3: "Task cảnh báo" - Chạy trên CORE 1
// ================================================================
void TaskCanhBao(void* pvParameters) {
  Serial.printf("[RTOS] Task canh bao khoi chay tren Core %d\n", xPortGetCoreID());

  float levelM = 0.0f;

  for (;;) {
    if (xQueueReceive(dataAlert, &levelM, portMAX_DELAY) == pdPASS) {
      if (!autoMode) {
        vTaskDelay(pdMS_TO_TICKS(200));
        continue;
      }

      if (levelM >= 1.0f && levelM < 1.5f) {
        alertRuleText = "Muc nuoc 1.0m-1.5m: Bat Bom 50% (10s)";
        Serial.printf("[Core 1 - Alert] Muc nuoc = %.2fm (1.0m - 1.5m) -> BAT MAY BOM 50%% trong 10s\n", levelM);
        setActuators(50, false);

        for (int i = 0; i < 50 && autoMode; i++) {
          vTaskDelay(pdMS_TO_TICKS(200));
        }
        if (autoMode) {
          setActuators(0, false);
          alertRuleText = "Hoan thanh chu ky Bom 50% (10s)";
          Serial.println("[Core 1 - Alert] Da het 10s -> TAT MAY BOM (50%)");
        }
      }
      else if (levelM >= 1.5f && levelM <= 2.0f) {
        alertRuleText = "Muc nuoc 1.5m-2.0m: Bat Bom 100% (20s)";
        Serial.printf("[Core 1 - Alert] Muc nuoc = %.2fm (1.5m - 2.0m) -> BAT MAY BOM 100%% trong 20s\n", levelM);
        setActuators(100, false);

        for (int i = 0; i < 100 && autoMode; i++) {
          vTaskDelay(pdMS_TO_TICKS(200));
        }
        if (autoMode) {
          setActuators(0, false);
          alertRuleText = "Hoan thanh chu ky Bom 100% (20s)";
          Serial.println("[Core 1 - Alert] Da het 20s -> TAT MAY BOM (100%)");
        }
      }
      else if (levelM < 0.5f) {
        alertRuleText = "Muc nuoc < 0.5m: Mo Van thoat (10s)";
        Serial.printf("[Core 1 - Alert] Muc nuoc = %.2fm (< 0.5m) -> BAT VAN THOAT trong 10s\n", levelM);
        setActuators(0, true);

        for (int i = 0; i < 50 && autoMode; i++) {
          vTaskDelay(pdMS_TO_TICKS(200));
        }
        if (autoMode) {
          setActuators(0, false);
          alertRuleText = "Hoan thanh chu ky Van thoat (10s)";
          Serial.println("[Core 1 - Alert] Da het 10s -> DONG VAN THOAT");
        }
      }
      else {
        alertRuleText = "Muc nuoc an toan (0.5m - 1.0m)";
        setActuators(0, false);
      }
    }
  }
}

// ================================================================
// 11. KHỞI TẠO (SETUP) & LOOP
// ================================================================
void setup() {
  Serial.begin(115200);
  delay(500);
  Serial.println("\n=======================================================");
  Serial.println("HE THONG RTOS GIAM SAT MOI TRUONG BE NUOI THUY HAI SAN");
  Serial.println("=======================================================");

  pinMode(TRIG_PIN, OUTPUT);
  pinMode(ECHO_PIN, INPUT);
  pinMode(PUMP_PIN, OUTPUT);
  pinMode(VALVE_PIN, OUTPUT);
  setActuators(0, false);

  dht.setup(DHT_PIN, DHTesp::DHT22);

  dataQueue = xQueueCreate(1, sizeof(SensorData));
  dataAlert = xQueueCreate(1, sizeof(float));

  xTaskCreatePinnedToCore(TaskTruyenData, "TaskTruyenData", 8192, NULL, 2, NULL, 0);
  xTaskCreatePinnedToCore(TaskDocData,    "TaskDocData",    4096, NULL, 2, NULL, 1);
  xTaskCreatePinnedToCore(TaskCanhBao,    "TaskCanhBao",    4096, NULL, 1, NULL, 1);
}

void loop() {
  vTaskDelete(NULL);
}
